{{- /* Renders the tech stack of a page as monochrome inline SVGs. Usage: {{ partial "tech-icons.html" .Params.tech }} Each entry is looked up in data/tech.yaml for a label + icon name, then resolved to assets/icons/tech/.svg. Unknown entries fall back to a text chip so nothing silently disappears. */ -}} {{- with . }} {{- range . }} {{- $key := . | lower }} {{- $meta := index (default dict site.Data.tech) $key }} {{- $label := . }} {{- $icon := $key | urlize }} {{- with $meta }} {{- with .label }}{{ $label = . }}{{ end }} {{- with .icon }}{{ $icon = . }}{{ end }} {{- end }} {{- $res := resources.Get (printf "icons/tech/%s.svg" $icon) }} {{- if $res }} {{- $svg := $res.Content | replaceRE `.*?` "" | replaceRE `\srole="img"` "" | replaceRE `{{ $svg | safeHTML }} {{- else }} {{ $label }} {{- end }} {{- end }} {{- end }}
